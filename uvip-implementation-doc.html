import 'package:google_mlkit_image_labeling/google_mlkit_image_labeling.dart';

class AiCamProvider with ChangeNotifier {
  ImageLabeler? _labeler;
  List<DetectedLabel> _labels = [];
  bool _isBusy = false;

  // Label urban yang ingin ditampilkan sebagai bounding area
  static const _urbanLabels = {
    'Building', 'Road', 'Sky', 'Tree',
    'Sidewalk', 'Vehicle', 'Wall', 'Vegetation'
  };

  void initLabeler() {
    _labeler = ImageLabeler(
      options: ImageLabelerOptions(confidenceThreshold: 0.55),
    );
  }

  /// Dipanggil dari startImageStream di AiCamScreen
  Future<void> processFrame(CameraImage image, InputImageRotation rotation) async {
    if (_isBusy || _labeler == null) return;
    _isBusy = true;

    try {
      final inputImage = InputImage.fromBytes(
        bytes: image.planes[0].bytes,
        metadata: InputImageMetadata(
          size: Size(image.width.toDouble(), image.height.toDouble()),
          rotation: rotation,
          format: InputImageFormat.nv21,
          bytesPerRow: image.planes[0].bytesPerRow,
        ),
      );

      final results = await _labeler!.processImage(inputImage);

      // Filter hanya label urban yang relevan
      _labels = results
          .where((l) => _urbanLabels.contains(l.label) && l.confidence > 0.55)
          .toList();

      notifyListeners();
    } finally {
      _isBusy = false;
    }
  }

  List<DetectedLabel> get labels => _labels;

  void dispose() {
    _labeler?.close();
    super.dispose();
  }
}

import asyncio
import httpx
from app.core.config import settings

async def _trigger_ai_processing(photo_id: str, file_path: str):
    """Fire-and-forget: kirim foto ke uvip-ai untuk diproses."""
    try:
        async with httpx.AsyncClient(timeout=120.0) as client:
            with open(file_path, "rb") as f:
                await client.post(
                    f"{settings.UVIP_AI_URL}/ai/process",
                    files={"file": (Path(file_path).name, f, "image/jpeg")},
                    data={"photo_id": photo_id},
                    headers={"X-Internal-Token": settings.UVIP_AI_TOKEN},
                )
    except Exception as e:
        logger.warning("AI trigger gagal photo_id=%s: %s", photo_id, e)
        # Tidak raise — upload tetap sukses, AI proses belakangan

# Di dalam endpoint upload foto, setelah db.commit():

from fastapi import File, Form, UploadFile, HTTPException, BackgroundTasks
import httpx
from uvip_ai.config import settings

@app.post("/ai/process")
async def process_image(
    file: UploadFile = File(...),
    photo_id: str = Form(""),  # ID foto dari backend
    background_tasks: BackgroundTasks = BackgroundTasks(),
):
    start = time.time()
    path = Path(f"tmp_{photo_id}_{file.filename}")

    try:
        # Simpan file sementara
        with open(path, "wb") as buf:
            shutil.copyfileobj(file.file, buf)

        img = cv2.imread(str(path))

        # 1. Privacy masking
        masked = privacy_guard.process_image(img)

        # 2. Segmentasi → 5 metrik
        seg_result = segformer.infer(masked["masked_image"])
        metrics = seg_result["metrics"]

        # 3. Prediksi skor (skip DINOv2 sementara jika RAM terbatas)
        scores = predictor.predict(metrics)  # input = 5 metrik saja

        result = {
            "photo_id": photo_id,
            "green_coverage_pct": metrics["green_coverage_pct"],
            "building_coverage_pct": metrics["building_coverage_pct"],
            "walkability_ratio": metrics["walkability_ratio"],
            "visual_clutter_index": metrics["visual_clutter_index"],
            "sky_visibility_pct": metrics["sky_visibility_pct"],
            "beauty_score": scores["beauty"],
            "safety_score": scores["safety"],
            "comfort_score": scores["comfort"],
            "uvi_score": scores["uvi"],
        }

        # 4. Callback ke backend-uvip (background, tidak blocking)
        if photo_id:
            background_tasks.add_task(_post_result_to_backend, result)

        elapsed = (time.time() - start) * 1000
        logger.info("✅ %s — %.0fms", photo_id, elapsed)
        return JSONResponse(result)

    finally:
        path.unlink(missing_ok=True)


async def _post_result_to_backend(result: dict):
    """Kirim hasil segmentasi ke backend-uvip untuk disimpan ke DB."""
    try:
        async with httpx.AsyncClient(timeout=30.0) as client:
            await client.post(
                f"{settings.uvip_api_base_url}/segmentation-results/",
                json=result,
                headers={"Authorization": f"Bearer {settings.uvip_api_token}"},
            )
    except Exception as e:
        logger.error("Callback ke backend gagal: %s", e)

import 'package:uvip/services/result_service.dart';
import 'package:uvip/models/segmentation_result_model.dart';

class ResultProvider with ChangeNotifier {
  SegmentationResultModel? _segResult;
  SegmentationResultModel? get segResult => _segResult;

  bool _isPolling = false;
  bool get isPolling => _isPolling;

  String? _processingStatus;  // "processing" | "done" | "timeout"
  String? get processingStatus => _processingStatus;

  /// Dipanggil setelah upload foto berhasil
  Future<void> pollSegmentationResult(String photoId) async {
    _isPolling = true;
    _processingStatus = 'processing';
    _segResult = null;
    notifyListeners();

    // Poll maksimal 30x dengan interval 2 detik = timeout 60 detik
    for (int i = 0; i < 30; i++) {
      await Future.delayed(const Duration(seconds: 2));

      try {
        final result = await ResultService().getSegmentationByPhotoId(photoId);
        if (result != null) {
          _segResult = result;
          _processingStatus = 'done';
          _isPolling = false;
          notifyListeners();
          return;
        }
      } catch (_) {
        // Lanjut polling meski ada error sementara
      }
    }

    // Timeout setelah 60 detik
    _processingStatus = 'timeout';
    _isPolling = false;
    notifyListeners();
  }
}

Future<SegmentationResultModel?> getSegmentationByPhotoId(String photoId) async {
  final response = await _dio.get(
    '/segmentation-results/',
    queryParameters: {'photo_id': photoId},
  );
  if (response.data is List && (response.data as List).isNotEmpty) {
    return SegmentationResultModel.fromJson(response.data[0]);
  }
  return null;  // belum ada → lanjut polling
}

Deploy Backend + AI dalam 1 VPS
Konfigurasi docker-compose.yml untuk gabungkan backend-uvip dan uvip-ai
docker-compose.yml — 1 VPS, 2 service, 1 network
version: '3.9'

services:
  # ── Database ──────────────────────────────────────
  db:
    image: postgis/postgis:15-3.3
    environment:
      POSTGRES_DB: uvip
      POSTGRES_USER: uvip
      POSTGRES_PASSWORD: ${DB_PASSWORD}
    volumes:
      - pgdata:/var/lib/postgresql/data
    restart: always

  # ── Backend API (port 8000) ────────────────────────
  backend:
    build: ./backend-uvip
    ports:
      - "8000:8000"
    environment:
      DATABASE_URL: postgresql://uvip:${DB_PASSWORD}@db:5432/uvip
      SECRET_KEY: ${SECRET_KEY}
      UVIP_AI_URL: http://uvip-ai:8001  # nama service, bukan localhost
    depends_on:
      - db
    volumes:
      - ./uploads:/app/uploads
    restart: always

  # ── AI Service (port 8001, internal only) ─────────
  uvip-ai:
    build: ./uvip-ai
    ports:
      - "127.0.0.1:8001:8001"  # hanya accessible dari dalam VPS
    environment:
      UVIP_API_BASE_URL: http://backend:8000
      UVIP_API_TOKEN: ${INTERNAL_TOKEN}
      UVIP_DEVICE: cpu
      UVIP_LOW_VRAM_MODE: "true"
      UVIP_USE_FP16: "false"  # fp16 tidak ada untungnya di CPU
    volumes:
      - ./uvip-ai/models:/app/models   # model tidak re-download tiap restart
      - ./uvip-ai/data:/app/data
    deploy:
      resources:
        limits:
          memory: 3G   # batas RAM agar tidak OOM seluruh VPS
    restart: always

volumes:
  pgdata:
